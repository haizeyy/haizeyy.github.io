No actual payment has been made.' ); } function copyCode() { const code = 'YARNER10'; const message = document.getElementById('copy-message'); if(navigator.clipboard && window.isSecureContext) { navigator.clipboard.writeText(code).then(() => { message.textContent = 'Discount code copied: ' + code; }).catch(() => { message.textContent = 'Your discount code: ' + code; }); } else { message.textContent = 'Your discount code: ' + code; } }
